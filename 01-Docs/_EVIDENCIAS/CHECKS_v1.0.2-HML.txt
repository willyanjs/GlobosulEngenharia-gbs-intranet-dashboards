## VALIDAÇÃO #7: Verificar ownership (segurança)

psql "$DATABASE_URL" -c "
SELECT 
  n.nspname AS schema,
  c.relname AS object_name,
  c.relkind AS type,
  pg_catalog.pg_get_userbyid(c.relowner) AS owner
FROM pg_class c
JOIN pg_namespace n ON n.oid = c.relnamespace
WHERE n.nspname = 'operacional'
  AND c.relname IN ('obras', 'update_obras_timestamp')
ORDER BY c.relname;
"

# Resultado esperado:
# operacional | obras | r (table) | gbs_dev
# operacional | update_obras_timestamp | p (function) | gbs_dev
