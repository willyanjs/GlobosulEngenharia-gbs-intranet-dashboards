SCRIPTS DE VALIDAÇÃO
=====================================================

# 1) Aplicar migrations completas
export DATABASE_URL="host=192.168.1.92 port=5432 dbname=gbs_intranet_prod user=gbs_dev password='...'"
cd 03-DB
bash apply_migrations.sh

# 2) Verificar faixa detectada
grep -E "Iniciando aplicação das migrations \(ordem:" logs/migrations.log
# Esperado: "ordem: 000 → 016"

# 3) Verificar índices de audit.auditoria
psql "$DATABASE_URL" -c "\d audit.auditoria" | grep -i index

# 4) Verificar índices de auth.reset_tokens (após 016)
psql "$DATABASE_URL" -c "
SELECT indexname 
FROM pg_indexes 
WHERE schemaname = 'auth' AND tablename = 'reset_tokens'
ORDER BY indexname;
"
# Esperado: idx_reset_tokens_user_id, _token, _expires_at, _ativos

# 5) Smoke test
psql "$DATABASE_URL" -c "
INSERT INTO auth.reset_tokens(id_usuario, token, expira_em)
SELECT id_usuario, 'TEST-' || gen_random_uuid(), now() + interval '1 hour'
FROM auth.usuarios LIMIT 1;

SELECT count(*) FROM auth.reset_tokens 
WHERE usado_em IS NULL AND revogado_em IS NULL AND expira_em > now();
"
